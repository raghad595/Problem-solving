void DFS(int currentNode, int searchedNode, vector<vector<int>> graph, bool visited[]) {
	visited[currentNode] = true;
	if (currentNode == searchedNode) return;
	cout << currentNode << " -> ";

	for (int i = 0; i < graph[currentNode].size(); i++) {
		int child = graph[currentNode][i];
		if (!visited[child])
			DFS(child, searchedNode, graph, visited);
	}
}

void BFS(int currentNode, int searchedNode, vector<vector<int>> graph, bool visited[]) {
	visited[currentNode] = true;
	if (currentNode == searchedNode) return;
	queue<int> q;
	q.push(currentNode);

	while (!q.empty())	{
		currentNode = q.front();
		cout << q.front() << " -> ";
		for (int i = 0; i < graph[currentNode].size(); i++) {
			int child = graph[currentNode][i];

			if (!visited[child]) {
				q.push(child);
				visited[child] = true;
			}
		}
		q.pop();
	}
}

//Shortest Path Algorithm Using BFS in Unweighted Graph
//======================================================
void shortestPath(int currentNode, int searchedNode, vector<vector<int>> graph, bool visited[]) {
	int start = currentNode;
	visited[currentNode] = true;
	queue<pair<int, int>> q;
	q.push({ currentNode, 0 });
	int  distance, child;
	while (!q.empty()){
		currentNode = q.front().first;
		distance = q.front().second;
		if (currentNode == searchedNode) {
			cout << "The shortest path between ("
				 << start << ") and (" << searchedNode
				 << ") = " << distance << endl;
			return;
		}
		for (int i = 0; i < graph[currentNode].size(); i++) {
			child = graph[currentNode][i];
			if (!visited[child]) {
				q.push({ child, distance + 1 });
				visited[child] = true;
			}
		}
		q.pop();
	}
}

//Path Using DFS will not be the shortest one
//======================================================
void pathByDFS(int currentNode, int searchedNode, vector<vector<int>> graph, bool visited[],int distance) {
	visited[currentNode] = true;
	if (currentNode == searchedNode) {
		cout << "The path = " << distance << endl;
		return;
	}

	for (int i = 0; i < graph[currentNode].size(); i++) {
		int child = graph[currentNode][i];
		if (!visited[child])
			pathByDFS(child, searchedNode, graph, visited, distance + 1);
	}
}
