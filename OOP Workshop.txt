class Duration {
	int hours, minutes, seconds;
public:
	Duration(){
		hours = minutes = seconds = 0;
	}
	Duration(int hours, int minutes, int seconds) {
		this->hours = hours;
		this->minutes = minutes;
		this->seconds = seconds;
	}

	Duration(int millisecond) {
		seconds = millisecond / 1000;
		hours = seconds / 3600;
		minutes = (seconds % 3600) / 60;
		seconds = seconds % 3600 % 60;
	}
	Duration(Duration& d) {
		this->hours = d.hours;
		this->minutes = d.minutes;
		this->seconds = d.seconds;
	}

	int converttosecond(Duration d) {
		return d.hours * 3600 + d.minutes * 60 + d.seconds;
	}

	Duration converttotime(int sum) {
		int newhours, newminutes, newseconds;
		newhours = sum / 3600;
		newminutes = sum % 3600 / 60;
		newseconds = sum % 3600 % 60;
		Duration newtime(newhours, newminutes, newseconds);
		return newtime;
	}

	Duration operator +(Duration d) {
		/*int newhours, newminutes, newseconds;
		newhours = this->hours + d.hours;
		newminutes = this->minutes + d.minutes;
		newseconds = this->seconds + d.seconds;*/
		int converted1 = converttosecond(*this);
		int converted2 = converttosecond(d);
		int sum = converted1 + converted2;

		Duration temp;
		temp = converttotime(sum);
		return temp;
	}

	Duration operator -(Duration d) {
		
		int converted1 = converttosecond(*this);
		int converted2 = converttosecond(d);
		int sum = converted1 - converted2;

		Duration temp;
		temp = converttotime(sum);
		return temp;
	}

	bool operator > (Duration d) {
		int converted1 = converttosecond(*this);
		int converted2 = converttosecond(d);
		return converted1 > converted2;
	}

	bool operator < (Duration d) {
		int converted1 = converttosecond(*this);
		int converted2 = converttosecond(d);
		return converted1 < converted2;
	}

	bool operator ==(Duration d) {
		return (this->hours == d.hours && this->minutes == d.minutes && this->seconds == d.seconds);
	}

	void Display() {
		cout << hours << " : " << minutes << " : " << seconds << endl;
	}
};


int main() {
	Duration d1(4, 50, 55), d2(4, 50, 55);
	Duration d3;

	if (d1 == d2) {
		cout << "true";
	}
	else {
		cout << "false";
	}
	
}
