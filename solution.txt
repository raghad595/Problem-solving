int primearrsize = 0;
int* Prime(int arr[], int size) {
    int count;
    int* primenumbers = new int[size];

    for (int i = 0; i < size; i++) {
        count = 0;
        for (int j = 1; j <= arr[i]; j++) {
            if (arr[i] % j == 0) {
                count++;
            }
        }
        if (count == 2) {
            primenumbers[primearrsize] = arr[i];
            primearrsize++;
        }
    }
    return primenumbers;
}

int main()
{
    int arr[10];
    for (int i = 0; i < 10; i++) {
        cin >> arr[i];
    }

    int* p = Prime(arr, 10);
    
    cout << "asd";

    for (int k = 0; k < primearrsize; k++) {
        cout << p[k] << endl;
    }
}
